Ilosc klientów wg płciSELECT IIf([plec]="K","Female","Male") AS Płeć, Count(Klienci.id_klienta) AS Ilość FROM Klienci GROUP BY IIf([plec]="K","Female","Male"); Spring Data Repository Query
Ilosc produktów wg dostepnosciSELECT IIf([dostepny],"Dostępny","Niedostępny") AS Dostępność, Count(Produkty.id_produktu) AS Ilość FROM Produkty GROUP BY IIf([dostepny],"Dostępny","Niedostępny"); Spring Data Repository Query
Kategorie wg lat zamówieńSELECT Year([data]) AS rok, Kategorie.nazwa, Sum(Produkty_zamowienia.ilosc) AS [Ilosc produktów] FROM Zamowienia INNER JOIN ((Produkty INNER JOIN (Kategorie INNER JOIN Produkty_Kategorie ON Kategorie.id_kategorii = Produkty_Kategorie.id_kategorii) ON Produkty.id_produktu = Produkty_Kategorie.id_produktu) INNER JOIN Produkty_zamowienia ON Produkty.id_produktu = Produkty_zamowienia.id_produktu) ON Zamowienia.id_zamowienia = Produkty_zamowienia.id_zamowienia GROUP BY Year([data]), Kategorie.nazwa; Spring Data Repository Query
Kategorie wg lat zamówień cenySELECT Year([data]) AS rok, Kategorie.nazwa, Sum([ilosc]*[cena_jednostkowa]) AS [Wartość zamówień] FROM Zamowienia INNER JOIN ((Produkty INNER JOIN (Kategorie INNER JOIN Produkty_Kategorie ON Kategorie.id_kategorii = Produkty_Kategorie.id_kategorii) ON Produkty.id_produktu = Produkty_Kategorie.id_produktu) INNER JOIN Produkty_zamowienia ON Produkty.id_produktu = Produkty_zamowienia.id_produktu) ON Zamowienia.id_zamowienia = Produkty_zamowienia.id_zamowienia GROUP BY Year([data]), Kategorie.nazwa; Spring Data Repository Query
Kategorie wg lat zamówień ceny_KrzyżowaTRANSFORM Sum([Kategorie wg lat zamówień ceny].[Wartość zamówień]) AS [SumaOfWartość zamówień] SELECT [Kategorie wg lat zamówień ceny].rok AS Rok, Sum([Kategorie wg lat zamówień ceny].[Wartość zamówień]) AS [Suma wartości zamówień] FROM [Kategorie wg lat zamówień ceny] GROUP BY [Kategorie wg lat zamówień ceny].rok PIVOT [Kategorie wg lat zamówień ceny].nazwa; Spring Data Repository Query
Kategorie wg lat zamówień_KrzyżowaTRANSFORM Sum([Kategorie wg lat zamówień].[Ilosc produktów]) AS Ilość SELECT [Kategorie wg lat zamówień].rok AS Rok, Sum([Kategorie wg lat zamówień].[Ilosc produktów]) AS [Łączna ilość produktów] FROM [Kategorie wg lat zamówień] GROUP BY [Kategorie wg lat zamówień].rok ORDER BY [Kategorie wg lat zamówień].rok PIVOT [Kategorie wg lat zamówień].nazwa; Spring Data Repository Query
Klienci wydający najwięcejSELECT TOP 10 [imie]+" "+[nazwisko] AS [Imię i nazwisko], Sum(Zamowienia.cena) AS [Suma wydatków] FROM Klienci INNER JOIN Zamowienia ON Klienci.id_klienta = Zamowienia.id_klienta GROUP BY [imie]+" "+[nazwisko], Klienci.id_klienta ORDER BY Sum(Zamowienia.cena) DESC; Spring Data Repository Query
Miasta z największą ilością zamówieńSELECT TOP 15 Adresy.miasto AS Miasto, Count(Zamowienia.id_zamowienia) AS [Ilosc zamowien] FROM (Klienci INNER JOIN Adresy ON Klienci.id_klienta = Adresy.id_klienta) INNER JOIN (Zamowienia INNER JOIN (Produkty INNER JOIN Produkty_zamowienia ON Produkty.id_produktu = Produkty_zamowienia.id_produktu) ON Zamowienia.id_zamowienia = Produkty_zamowienia.id_zamowienia) ON Klienci.id_klienta = Zamowienia.id_klienta GROUP BY Adresy.miasto ORDER BY Count(Zamowienia.id_zamowienia) DESC; Spring Data Repository Query
NieuzywaneKategorieSELECT Kategorie.id_kategorii, Kategorie.nazwa, Count(Produkty.id_produktu) AS Uzycia FROM Produkty RIGHT JOIN (Kategorie LEFT JOIN Produkty_Kategorie ON Kategorie.id_kategorii = Produkty_Kategorie.id_kategorii) ON Produkty.id_produktu = Produkty_Kategorie.id_produktu GROUP BY Kategorie.id_kategorii, Kategorie.nazwa HAVING (((Count(Produkty.id_produktu))=0)) ORDER BY Kategorie.id_kategorii; Spring Data Repository Query
Produkty generujące największe przychodySELECT Produkty.nazwa, Sum([ilosc]*[Zamowienia.cena]) AS Przychody FROM Zamowienia INNER JOIN (Produkty INNER JOIN Produkty_zamowienia ON Produkty.id_produktu = Produkty_zamowienia.id_produktu) ON Zamowienia.id_zamowienia = Produkty_zamowienia.id_zamowienia GROUP BY Produkty.id_produktu, Produkty.nazwa ORDER BY Sum([ilosc]*[Zamowienia.cena]) DESC; Spring Data Repository Query
Produkty w koszykuSELECT Koszyk.id_klienta, Produkty.nazwa, Produkty.opis, Produkty.cena FROM Produkty INNER JOIN Koszyk ON Produkty.id_produktu = Koszyk.id_produktu; Spring Data Repository Query
